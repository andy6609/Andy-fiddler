cmake_minimum_required(VERSION 3.20)
project(WireScope VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h REQUIRED)
find_library(NGHTTP2_LIBRARY NAMES nghttp2 REQUIRED)
# src/http2.cpp uses the nghttp2_ssize API (1.60+). The older ssize_t API does not compile with MSVC, which has no ssize_t.
file(STRINGS ${NGHTTP2_INCLUDE_DIR}/nghttp2/nghttp2ver.h nghttp2_version_line REGEX "#define NGHTTP2_VERSION \"")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" NGHTTP2_VERSION "${nghttp2_version_line}")
if(NGHTTP2_VERSION VERSION_LESS 1.60.0)
  message(FATAL_ERROR "nghttp2 ${NGHTTP2_VERSION} found; WireScope needs 1.60 or newer (nghttp2_ssize API)")
endif()
add_library(wirescope_core src/core.cpp src/packet.cpp)
target_include_directories(wirescope_core PUBLIC src third_party)
target_link_libraries(wirescope_core PUBLIC OpenSSL::Crypto Threads::Threads)
add_executable(WireScope src/main.cpp src/capture.cpp src/proxy.cpp src/tls.cpp src/http2.cpp src/process.cpp src/sysproxy.cpp)
target_include_directories(WireScope PRIVATE ${NGHTTP2_INCLUDE_DIR})
target_link_libraries(WireScope PRIVATE wirescope_core OpenSSL::SSL ${NGHTTP2_LIBRARY})
if(WIN32)
  target_compile_definitions(wirescope_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(WireScope PRIVATE ws2_32 crypt32 shell32 advapi32 iphlpapi wininet)
else()
  find_library(PCAP_LIBRARY pcap REQUIRED)
  target_link_libraries(WireScope PRIVATE ${PCAP_LIBRARY})
endif()
if(MSVC)
  target_compile_options(wirescope_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(wirescope_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
# Copy UI assets on every build, not only when the executable relinks.
add_custom_target(wirescope_web ALL COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web $<TARGET_FILE_DIR:WireScope>/web)
add_dependencies(WireScope wirescope_web)
option(WIRESCOPE_BUILD_LAB "Build the lab/ learning versions" ON)
if(WIRESCOPE_BUILD_LAB)
  add_subdirectory(lab)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(wirescope_tests tests/core_tests.cpp)
  target_link_libraries(wirescope_tests PRIVATE wirescope_core)
  target_compile_definitions(wirescope_tests PRIVATE WIRESCOPE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME core COMMAND wirescope_tests)
  # End-to-end tests: real curl -> WireScope (or lab/) -> local HTTP/HTTPS/HTTP-2 servers.
  # HTTP/2 upstream: nghttpd when installed (macOS/Linux), otherwise tests/h2_server.py on the
  # Python 'h2' package (pip install h2) — the Windows path, since nghttpd has no Windows build.
  find_package(Python3 COMPONENTS Interpreter)
  find_program(NGHTTPD_EXECUTABLE nghttpd HINTS /opt/homebrew/opt/nghttp2/bin /usr/local/opt/nghttp2/bin)
  find_program(CURL_EXECUTABLE curl)
  find_program(OPENSSL_EXECUTABLE openssl)
  set(test_env "CURL=${CURL_EXECUTABLE}" "OPENSSL=${OPENSSL_EXECUTABLE}" "PYTHONDONTWRITEBYTECODE=1")
  set(h2_server_found FALSE)
  if(NGHTTPD_EXECUTABLE)
    set(h2_server_found TRUE)
    list(APPEND test_env "NGHTTPD=${NGHTTPD_EXECUTABLE}")
  elseif(Python3_Interpreter_FOUND)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import h2" RESULT_VARIABLE h2_import OUTPUT_QUIET ERROR_QUIET)
    if(h2_import EQUAL 0)
      set(h2_server_found TRUE)
    endif()
  endif()
  if(Python3_Interpreter_FOUND AND CURL_EXECUTABLE AND OPENSSL_EXECUTABLE AND h2_server_found)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py $<TARGET_FILE_DIR:WireScope>)
    set_tests_properties(integration PROPERTIES ENVIRONMENT "${test_env}" TIMEOUT 240)
    if(WIRESCOPE_BUILD_LAB)
      add_test(NAME lab COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/lab_smoke.py ${CMAKE_BINARY_DIR})
      set_tests_properties(lab PROPERTIES ENVIRONMENT "${test_env}" TIMEOUT 300)
    endif()
    if(WIN32)
      # Windows-only features (process attribution, system proxy) against real Windows APIs.
      add_test(NAME windows COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/windows_test.py $<TARGET_FILE_DIR:WireScope>)
      set_tests_properties(windows PROPERTIES ENVIRONMENT "${test_env}" TIMEOUT 240)
    endif()
  else()
    message(STATUS "integration/lab tests skipped: need Python 3, curl, openssl, and nghttpd or Python h2 (pip install h2)")
  endif()
endif()
install(TARGETS WireScope RUNTIME DESTINATION .)
install(DIRECTORY web DESTINATION .)
