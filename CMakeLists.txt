cmake_minimum_required(VERSION 3.20)
project(WireScope VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(NGHTTP2_INCLUDE_DIR nghttp2/nghttp2.h REQUIRED)
find_library(NGHTTP2_LIBRARY NAMES nghttp2 REQUIRED)
add_library(wirescope_core src/core.cpp src/packet.cpp)
target_include_directories(wirescope_core PUBLIC src third_party)
target_link_libraries(wirescope_core PUBLIC OpenSSL::Crypto Threads::Threads)
add_executable(WireScope src/main.cpp src/capture.cpp src/proxy.cpp src/tls.cpp src/http2.cpp)
target_include_directories(WireScope PRIVATE ${NGHTTP2_INCLUDE_DIR})
target_link_libraries(WireScope PRIVATE wirescope_core OpenSSL::SSL ${NGHTTP2_LIBRARY})
if(WIN32)
  target_compile_definitions(wirescope_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(WireScope PRIVATE ws2_32 crypt32 shell32 advapi32)
else()
  find_library(PCAP_LIBRARY pcap REQUIRED)
  target_link_libraries(WireScope PRIVATE ${PCAP_LIBRARY})
endif()
if(MSVC)
  target_compile_options(wirescope_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(wirescope_core PUBLIC -Wall -Wextra -Wpedantic)
endif()
# Copy UI assets on every build, not only when the executable relinks.
add_custom_target(wirescope_web ALL COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/web $<TARGET_FILE_DIR:WireScope>/web)
add_dependencies(WireScope wirescope_web)
include(CTest)
if(BUILD_TESTING)
  add_executable(wirescope_tests tests/core_tests.cpp)
  target_link_libraries(wirescope_tests PRIVATE wirescope_core)
  target_compile_definitions(wirescope_tests PRIVATE WIRESCOPE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME core COMMAND wirescope_tests)
  # End-to-end proxy test (curl + nghttpd + openssl). Development-host check; not run on Windows.
  find_package(Python3 COMPONENTS Interpreter)
  find_program(NGHTTPD_EXECUTABLE nghttpd HINTS /opt/homebrew/opt/nghttp2/bin /usr/local/opt/nghttp2/bin)
  find_program(CURL_EXECUTABLE curl)
  if(NOT WIN32 AND Python3_Interpreter_FOUND AND NGHTTPD_EXECUTABLE AND CURL_EXECUTABLE)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_test.py $<TARGET_FILE_DIR:WireScope>)
    set_tests_properties(integration PROPERTIES ENVIRONMENT "NGHTTPD=${NGHTTPD_EXECUTABLE}" TIMEOUT 180)
  endif()
endif()
install(TARGETS WireScope RUNTIME DESTINATION .)
install(DIRECTORY web DESTINATION .)
